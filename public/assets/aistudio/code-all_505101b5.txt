import { Fragment } from 'react';

/**
 * يعرض نصًّا فيه ترميز: [x] = الحرف الأصلي (برتقالي) ، {x} = الحرف بعد الإعلال (أخضر).
 * نكتفي باللون والخلفية دون هوامش حتى لا ينقطع اتصال الحروف العربية.
 */
export function Marked({ text, className = '' }: { text: string; className?: string }) {
  const parts = text.split(/(\[[^\]]+\]|\{[^}]+\})/g).filter((p) => p !== '');
  return (
    <span className={className}>
      {parts.map((p, i) => {
        if (p.startsWith('[') && p.endsWith(']'))
          return <span key={i} className="mk-orig">{p.slice(1, -1)}</span>;
        if (p.startsWith('{') && p.endsWith('}'))
          return <span key={i} className="mk-new">{p.slice(1, -1)}</span>;
        return <Fragment key={i}>{p}</Fragment>;
      })}
    </span>
  );
}

/** يزيل ترميز التمييز ويعيد النص العادي */
export function plain(text: string): string {
  return text.replace(/[[\]{}]/g, '');
}

export function ProgressBar({ value, max, label }: { value: number; max: number; label?: string }) {
  const pct = max > 0 ? Math.min(100, Math.round((value / max) * 100)) : 0;
  return (
    <div
      role="progressbar"
      aria-valuemin={0}
      aria-valuemax={max}
      aria-valuenow={value}
      aria-label={label}
      className="h-2.5 w-full overflow-hidden rounded-full bg-brand-100"
    >
      <div className="h-full rounded-full bg-brand-600 transition-all duration-500" style={{ width: `${pct}%` }} />
    </div>
  );
}

import type { QuizItem } from '../types';

export const LABELS = ['أ', 'ب', 'ج', 'د'];

interface Props {
  item: QuizItem;
  index: number;
  total: number;
  selected: number | null;
  confirmed: boolean;
  onSelect: (i: number) => void;
  onConfirm: () => void;
  onNext: () => void;
}

export function QuizQuestion({ item, index, total, selected, confirmed, onSelect, onConfirm, onNext }: Props) {
  const isCorrect = confirmed && selected === item.correctIndex;
  const last = index === total - 1;

  return (
    <div className="animate-rise space-y-5" key={item.id}>
      <div>
        <div className="mb-2 flex items-center justify-between text-sm font-bold text-slate-600">
          <span>السؤال {index + 1} من {total}</span>
          <span className="text-slate-400">{item.topic}</span>
        </div>
        <ProgressBar value={index + (confirmed ? 1 : 0)} max={total} label="تقدم الاختبار" />
      </div>

      <section className="rounded-3xl border border-brand-100 bg-white p-6 shadow-sm">
        <h2 className="text-2xl font-extrabold leading-10 text-slate-800">{item.text}</h2>

        <div role="radiogroup" aria-label="الخيارات" className="mt-5 space-y-3">
          {item.options.map((opt, i) => {
            const chosen = selected === i;
            const correct = confirmed && i === item.correctIndex;
            const wrong = confirmed && chosen && i !== item.correctIndex;
            let cls = 'border-slate-200 bg-white hover:border-brand-300 hover:bg-brand-50';
            if (!confirmed && chosen) cls = 'border-brand-600 bg-brand-50 ring-2 ring-brand-200';
            if (correct) cls = 'border-emerald-500 bg-emerald-50 ring-2 ring-emerald-200';
            if (wrong) cls = 'border-red-500 bg-red-50 ring-2 ring-red-200';
            if (confirmed && !correct && !wrong) cls = 'border-slate-200 bg-white opacity-60';
            return (
              <button
                key={opt}
                type="button"
                role="radio"
                aria-checked={chosen}
                disabled={confirmed}
                onClick={() => onSelect(i)}
                className={`flex w-full items-center gap-3 rounded-2xl border-2 px-4 py-3.5 text-right text-lg font-semibold text-slate-800 transition ${cls} ${confirmed ? 'cursor-default' : 'cursor-pointer'}`}

                <span
                  className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-base font-extrabold ${
                    correct ? 'bg-emerald-500 text-white' : wrong ? 'bg-red-500 text-white' : chosen ? 'bg-brand-600 text-white' : 'bg-brand-100 text-brand-700'

              </button>
            );
          })}

        {!confirmed && (
          <button
            type="button"
            onClick={onConfirm}
            disabled={selected === null}
            className="mt-6 w-full rounded-2xl bg-brand-700 px-6 py-3.5 text-lg font-extrabold text-white shadow-sm transition enabled:hover:bg-brand-800 disabled:cursor-not-allowed disabled:bg-slate-200 disabled:text-slate-400 sm:w-auto"
          >
            تأكيد الإجابة
          </button>
        )}
      </section>

      {confirmed && (
        <section
          aria-live="polite"
          className={`animate-pop rounded-3xl border-2 p-5 ${isCorrect ? 'border-emerald-300 bg-emerald-50' : 'border-red-300 bg-red-50'}`}
        >
          <p className={`flex items-center gap-2 text-xl font-extrabold ${isCorrect ? 'text-emerald-700' : 'text-red-700'}`}>
            {isCorrect ? <CheckCircle2 aria-hidden /> : <XCircle aria-hidden />}
            {isCorrect ? 'إجابة صحيحة، أحسنت!' : 'إجابة خاطئة'}

            {item.explanation}
          </p>

            type="button"
            onClick={onNext}
            className="mt-4 inline-flex items-center gap-2 rounded-2xl bg-brand-700 px-6 py-3 font-extrabold text-white shadow-sm transition hover:bg-brand-800"
          >
            {last ? 'عرض النتيجة' : 'السؤال التالي'} <ArrowLeft size={18} aria-hidden />
          </button>

      )}

  );
}

import { useEffect, useState } from 'react';

export type Route =
  | { name: 'home' }
  | { name: 'lessons'; lessonId?: string }
  | { name: 'quiz' };

export function parseHash(hash: string): Route {
  const parts = hash.replace(/^#\/?/, '').split('/').filter(Boolean);
  if (parts[0] === 'lessons') return { name: 'lessons', lessonId: parts[1] ? decodeURIComponent(parts[1]) : undefined };
  if (parts[0] === 'quiz') return { name: 'quiz' };
  return { name: 'home' };
}

export function navigate(path: string) {
  window.location.hash = path;
}

export function useHashRoute(): Route {
  const [route, setRoute] = useState<Route>(() => parseHash(window.location.hash));
  useEffect(() => {
    const onChange = () => {
      setRoute(parseHash(window.location.hash));
      window.scrollTo({ top: 0 });
    };
    window.addEventListener('hashchange', onChange);
    return () => window.removeEventListener('hashchange', onChange);
  }, []);
  return route;
}

import { useCallback, useReducer, useRef } from 'react';
import type { Question, QuizItem, Store } from '../types';
import { buildQuiz, pickQuestionIds, QUIZ_SIZE } from '../utils/quiz';

export type Phase = 'start' | 'question' | 'result' | 'review';

interface State {
  phase: Phase;
  items: QuizItem[];
  index: number;
  selected: number | null;
  confirmed: boolean;
  answers: (number | null)[];
}

type Action =
  | { type: 'start'; items: QuizItem[] }
  | { type: 'select'; i: number }
  | { type: 'confirm' }
  | { type: 'next' }
  | { type: 'phase'; phase: Phase }
  | { type: 'reset' };

const initial: State = { phase: 'start', items: [], index: 0, selected: null, confirmed: false, answers: [] };

function reducer(s: State, a: Action): State {
  switch (a.type) {
    case 'start':
      return { phase: 'question', items: a.items, index: 0, selected: null, confirmed: false, answers: a.items.map(() => null) };
    case 'select':
      if (s.phase !== 'question' || s.confirmed) return s; // لا يمكن تغيير الإجابة بعد التأكيد
      return { ...s, selected: a.i };
    case 'confirm': {
      if (s.phase !== 'question' || s.confirmed || s.selected === null) return s;
      const answers = s.answers.slice();
      answers[s.index] = s.selected;
      return { ...s, confirmed: true, answers };
    }
    case 'next':
      if (!s.confirmed) return s;
      if (s.index >= s.items.length - 1) return { ...s, phase: 'result' };
      return { ...s, index: s.index + 1, selected: null, confirmed: false };
    case 'phase':
      return { ...s, phase: a.phase };
    case 'reset':
      return initial;
    default:
      return s;
  }
}

export function scoreOf(items: QuizItem[], answers: (number | null)[]) {
  const correct = items.reduce((n, it, i) => n + (answers[i] === it.correctIndex ? 1 : 0), 0);
  return { correct, wrong: items.length - correct, total: items.length };
}

export function useQuiz(bank: Question[], store: Store, update: (fn: (s: Store) => Store) => void) {
  const [state, dispatch] = useReducer(reducer, initial);
  const savedRef = useRef(false);

  const canStart = bank.length >= QUIZ_SIZE;

  const start = useCallback(() => {
    if (bank.length < QUIZ_SIZE) return;
    const picked = pickQuestionIds(bank, store.seen, QUIZ_SIZE);
    const items = buildQuiz(bank, picked.ids);
    savedRef.current = false;
    update((s) => ({ ...s, seen: picked.seen, cycle: picked.newCycle ? s.cycle + 1 : s.cycle }));
    dispatch({ type: 'start', items });
  }, [bank, store.seen, update]);

  const next = useCallback(() => {
    // عند آخر سؤال: نحفظ النتيجة مرة واحدة فقط
    if (state.confirmed && state.index >= state.items.length - 1 && !savedRef.current) {
      savedRef.current = true;
      const { correct, total } = scoreOf(state.items, state.answers);
      update((s) => ({
        ...s,
        attempts: [...s.attempts, { id: `${Date.now()}`, date: new Date().toISOString(), score: correct, total }],
      }));
    }
    dispatch({ type: 'next' });
  }, [state, update]);

  const resetCycle = useCallback(() => {
    update((s) => ({ ...s, seen: [], cycle: s.cycle + 1 }));
  }, [update]);

  const clearHistory = useCallback(() => {
    update((s) => ({ ...s, attempts: [], seen: [], cycle: 1 }));
  }, [update]);

  return {
    state,
    canStart,
    start,
    next,
    resetCycle,
    clearHistory,
    select: (i: number) => dispatch({ type: 'select', i }),
    confirm: () => dispatch({ type: 'confirm' }),
    setPhase: (phase: Phase) => dispatch({ type: 'phase', phase }),
    reset: () => dispatch({ type: 'reset' }),
  };
}

import { useCallback, useState } from 'react';
import type { Store } from '../types';
import { loadStore, saveStore } from '../utils/storage';

/** حالة التقدّم المحلية (LocalStorage) — دون حسابات أو خوادم */
export function useStore() {
  const [store, setStore] = useState<Store>(() => loadStore());

  const update = useCallback((fn: (s: Store) => Store) => {
    setStore((prev) => {
      const next = fn(prev);
      saveStore(next);
      return next;
    });
  }, []);

  return { store, update };
}

import type { Question, QuizItem } from '../types';
import { randomInt, shuffle } from './random';

export const QUIZ_SIZE = 10;
export const OPTIONS_COUNT = 4;

/** يتحقق من سلامة بنك الأسئلة ويستبعد أي سؤال معيب مع تسجيل السبب */
export function validateBank(raw: unknown): { valid: Question[]; errors: string[] } {
  const errors: string[] = [];
  const valid: Question[] = [];
  if (!Array.isArray(raw)) return { valid, errors: ['ملف الأسئلة يجب أن يكون مصفوفة'] };
  const ids = new Set<string>();
  raw.forEach((q, i) => {
    const where = `السؤال رقم ${i + 1}`;
    if (!q || typeof q !== 'object') return errors.push(`${where}: صيغة غير صحيحة`), undefined;
    const o = q as Record<string, unknown>;
    if (typeof o.id !== 'string' || !o.id) return errors.push(`${where}: المعرّف مفقود`), undefined;
    if (ids.has(o.id)) return errors.push(`${where}: المعرّف ${o.id} مكرّر`), undefined;
    if (typeof o.text !== 'string' || !o.text.trim()) return errors.push(`${o.id}: نص السؤال مفقود`), undefined;
    if (!Array.isArray(o.options) || o.options.length !== OPTIONS_COUNT)
      return errors.push(`${o.id}: يجب أن يحتوي السؤال على ${OPTIONS_COUNT} خيارات`), undefined;
    if (o.options.some((x) => typeof x !== 'string' || !x.trim()))
      return errors.push(`${o.id}: أحد الخيارات فارغ`), undefined;
    if (new Set(o.options).size !== OPTIONS_COUNT) return errors.push(`${o.id}: خيارات مكرّرة`), undefined;
    if (!Number.isInteger(o.answer) || (o.answer as number) < 0 || (o.answer as number) >= OPTIONS_COUNT)
      return errors.push(`${o.id}: رقم الإجابة الصحيحة غير صالح`), undefined;
    if (typeof o.explanation !== 'string' || !o.explanation.trim())
      return errors.push(`${o.id}: شرح الإجابة مفقود`), undefined;
    ids.add(o.id);
    valid.push({
      id: o.id,
      topic: typeof o.topic === 'string' ? o.topic : undefined,
      text: o.text,
      options: o.options as string[],
      answer: o.answer as number,
      explanation: o.explanation,
    });
  });
  return { valid, errors };
}

/**
 * يختار معرّفات أسئلة الاختبار الجديد.
 * - لا تكرار داخل الاختبار الواحد.
 * - تُستبعد الأسئلة التي ظهرت في الدورة الحالية حتى تُستنفد.
 * - عند الاستنفاد تبدأ دورة جديدة بترتيب مختلف دون تكرار داخل الاختبار نفسه.
 */
export function pickQuestionIds(
  bank: readonly Question[],
  seen: readonly string[],
  count = QUIZ_SIZE,
): { ids: string[]; seen: string[]; newCycle: boolean } {
  const bankIds = bank.map((q) => q.id);
  const bankSet = new Set(bankIds);
  const seenValid = Array.from(new Set(seen.filter((id) => bankSet.has(id))));
  const seenSet = new Set(seenValid);
  const pool = shuffle(bankIds.filter((id) => !seenSet.has(id)));

  if (pool.length >= count) {
    const ids = pool.slice(0, count);
    return { ids: shuffle(ids), seen: [...seenValid, ...ids], newCycle: false };
  }
  // الأسئلة المتبقية كلها تدخل، ثم تبدأ دورة جديدة ونكمل منها
  const first = pool;
  const firstSet = new Set(first);
  const rest = shuffle(bankIds.filter((id) => !firstSet.has(id)));
  const fill = rest.slice(0, count - first.length);
  return { ids: shuffle([...first, ...fill]), seen: fill, newCycle: true };
}

function isAcceptable(a: number[]): boolean {
  for (let i = 0; i + 2 < a.length; i++) {
    if (a[i] === a[i + 1] && a[i + 1] === a[i + 2]) return false; // ثلاث متتالية متطابقة
  }
  for (let i = 0; i + 3 < a.length; i++) {
    const d1 = (a[i + 1] - a[i] + 4) % 4;
    const d2 = (a[i + 2] - a[i + 1] + 4) % 4;
    const d3 = (a[i + 3] - a[i + 2] + 4) % 4;
    if (d1 === d2 && d2 === d3 && (d1 === 1 || d1 === 3)) return false; // نمط أ ب ج د
  }
  return true;
}

/** مواقع الإجابات الصحيحة: موزّعة بالتساوي قدر الإمكان، مع تجنّب الأنماط المتكررة */
export function balancedPositions(n: number): number[] {
  for (let t = 0; t < 1000; t++) {
    const offset = randomInt(OPTIONS_COUNT);
    const arr = shuffle(Array.from({ length: n }, (_, i) => (i + offset) % OPTIONS_COUNT));
    if (isAcceptable(arr)) return arr;
  }
  return shuffle(Array.from({ length: n }, (_, i) => i % OPTIONS_COUNT));
}

/** يبني الاختبار: يخلط الخيارات مع الحفاظ على ارتباط الإجابة الصحيحة بنصّها */
export function buildQuiz(bank: readonly Question[], ids: readonly string[]): QuizItem[] {
  const byId = new Map(bank.map((q) => [q.id, q]));
  const positions = balancedPositions(ids.length);
  return ids.map((id, i) => {
    const q = byId.get(id);
    if (!q) throw new Error(`سؤال غير موجود: ${id}`);
    const correctText = q.options[q.answer];
    const wrong = shuffle(q.options.filter((_, idx) => idx !== q.answer));
    const options = wrong.slice();
    options.splice(positions[i], 0, correctText);
    return {
      id: q.id,
      topic: q.topic,
      text: q.text,
      options,
      correctIndex: positions[i],
      explanation: q.explanation,
    };
  });
}

/** عدد صحيح عشوائي في [0, max) باستخدام مولّد آمن إن توفّر */
export function randomInt(max: number): number {
  if (max <= 1) return 0;
  const c = typeof globalThis !== 'undefined' ? (globalThis as { crypto?: Crypto }).crypto : undefined;
  if (c && typeof c.getRandomValues === 'function') {
    const range = 0x100000000;
    const limit = range - (range % max);
    const buf = new Uint32Array(1);
    // رفض القيم الزائدة لتجنّب الانحياز
    for (let i = 0; i < 16; i++) {
      c.getRandomValues(buf);
      if (buf[0] < limit) return buf[0] % max;
    }
  }
  return Math.floor(Math.random() * max);
}

/** خلط فيشر-ييتس دون تعديل المصفوفة الأصلية */
export function shuffle<T>(input: readonly T[]): T[] {
  const a = input.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = randomInt(i + 1);
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

import type { Attempt, Store } from '../types';

const KEY = 'ilal:v1:store';
const MAX_ATTEMPTS = 100;

export function emptyStore(): Store {
  return { seen: [], cycle: 1, attempts: [], visited: [] };

function isStrArray(x: unknown): x is string[] {
  return Array.isArray(x) && x.every((v) => typeof v === 'string');
}

function isAttempt(x: unknown): x is Attempt {
  if (!x || typeof x !== 'object') return false;
  const a = x as Record<string, unknown>;
  return typeof a.id === 'string' && typeof a.date === 'string' && typeof a.score === 'number' && typeof a.total === 'number';
}

/** قراءة آمنة: أي خلل في التخزين (أو منعه) لا يكسر التطبيق */
export function loadStore(): Store {
  try {
    const raw = window.localStorage.getItem(KEY);
    if (!raw) return emptyStore();
    const o = JSON.parse(raw) as Record<string, unknown>;
    return {
      seen: isStrArray(o.seen) ? o.seen : [],
      cycle: typeof o.cycle === 'number' && o.cycle >= 1 ? o.cycle : 1,
      attempts: Array.isArray(o.attempts) ? o.attempts.filter(isAttempt) : [],
      visited: isStrArray(o.visited) ? o.visited : [],
    };
  } catch {
    return emptyStore();
  }
}

export function saveStore(s: Store): void {
  try {
    const trimmed: Store = { ...s, attempts: s.attempts.slice(-MAX_ATTEMPTS) };
    window.localStorage.setItem(KEY, JSON.stringify(trimmed));
  } catch {
    /* التخزين غير متاح (وضع خاص مثلًا): نتجاهل بهدوء */
  }
}
